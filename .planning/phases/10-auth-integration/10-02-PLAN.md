Add Amplify configuration (Cognito only):
```typescript

// {{/if AUTH_COGNITO}}
```

Update render to conditionally wrap with AuthProvider:
- When AUTH_COGNITO or AUTH_AUTH0 is true: Wrap App with AuthProvider inside ChakraProvider
- When neither is true: Just render App in ChakraProvider (current behavior)

Use nested conditionals:
```typescript
root.render(
  <StrictMode>
    <ChakraProvider theme={theme}>
      {/* {{#if AUTH_COGNITO}} */}
      <AuthProvider>
      {/* {{/if AUTH_COGNITO}} */}
      {/* {{#if AUTH_AUTH0}} */}
      <AuthProvider>
      {/* {{/if AUTH_AUTH0}} */}
        <App />
      {/* {{#if AUTH_COGNITO}} */}
      </AuthProvider>
      {/* {{/if AUTH_COGNITO}} */}
      {/* {{#if AUTH_AUTH0}} */}
      </AuthProvider>
      {/* {{/if AUTH_AUTH0}} */}
    </ChakraProvider>
  </StrictMode>
);
```

Note: The double AuthProvider tags are intentional - only one set will be kept based on provider selection.


Also update use-auth.ts to conditionally import AuthContext from correct provider:
```typescript
import { useContext } from 'react';
// {{#if AUTH_COGNITO}}
import { AuthContext } from './cognito-provider';
// {{/if AUTH_COGNITO}}
// {{#if AUTH_AUTH0}}
import { AuthContext } from './auth0-provider';
// {{/if AUTH_AUTH0}}
import type { AuthContextType } from '{{PACKAGE_SCOPE}}/common-types';

export function useAuth(): AuthContextType {
  const context = useContext(AuthContext);
  if (!context) {
    throw new Error('useAuth must be used within an AuthProvider');
  }
  return context;
}
```
